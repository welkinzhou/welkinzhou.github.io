# Graph

所有节点如下：

```python
NODE_REGISTRY: dict[str, NodeFn] = {
    "memory": memory_node,
    "intent": intent_node,
    "retrieval": retrieval_node,
    "generate": generate_node,
    "persist": persist_node,
    "title": title_node,
}

# Fixed order. Do not reorder without updating node contracts (memory before intent).
CHAT_STAGES: tuple[str, ...] = (
    "memory",
    "intent",
    "retrieval",
    "generate",
    "persist",
    "title",
)


def build_chat_graph() -> CompiledStateGraph[ChatState]:
    """Compile the fixed chat pipeline graph."""
    graph: StateGraph[ChatState] = StateGraph(ChatState)
    for name in CHAT_STAGES:
        graph.add_node(name, NODE_REGISTRY[name])

    graph.add_edge(START, CHAT_STAGES[0])
    for left, right in pairwise(CHAT_STAGES):
        graph.add_edge(left, right)
    graph.add_edge(CHAT_STAGES[-1], END)
    return graph.compile()
```

## Intent（意图识别）

意图识别，是指确定用户问讯方向。以客服举例，有些人比较礼貌，提问前会先发你好这种问候语。这种无关业务的问题，可以让模型按照闲聊回复。有些用户会比较急躁，会直接说，你们的产品怎么这么差呀，会不会做生意。这些话太过笼统，我们需要展开询问，是什么产品、遇到了什么问题。

意图识别，暂定一下几种问题类型：

```python
class IntentKind(StrEnum):
    """意图识别类型"""
    RULE_QA = "rule_qa"
    CHITCHAT = "chitchat"
    FACTUAL_QA = "factual_qa"
    CLARIFICATION = "clarification"
    COMPLEX_QA = "complex_qa"
```

`RULE_QA` 指强规则匹配，例如当前知识库功能说明、产品定位、FAQ等。如果直接问模型，模型可能把自己的信息说出来了。以前就有，cursor/cc 外接第三方模型DeepSeek，你问他是什么模型，直接就回答自己是DeepSeek了。出现在这种情况，会严重影响产品形象。`CHITCHAT` 指闲聊，例如你好等问候，这些无需走检索。`FACTUAL_QA` 指意图明确清晰的问题，例如查询库内某个产品的规格，这是有明确信息的。`CLARIFICATION` 只需要用户补充信息的，比如用户说看看订单，我们需要追问几个点，例如哪个时间段内的，什么状态的订单，检索、生成都不走，直接返回追问。`COMPLEX_QA` 只需要拆分步骤的，比如说用户说我想学习使用某个工具，拆分成工具是什么、快速入门、基础概念解释、进阶使用。

剩下判断就需要增加 LLM 了，看下大致实现：

```python
async def intent_node(state: ChatState) -> ChatState:
    """路由本轮：规则 FAQ → 无知识库闲聊 → 一次 LLM 结构化意图分析。"""
    ctx = state["ctx"]

    # 固定 FAQ：跳过 LLM 与检索，经 skip_generate 直接吐标准答。
    rule_answer = match_rule_answer(ctx.user_message)
    if rule_answer is not None:
        analysis = IntentAnalysis(
            kind=IntentKind.RULE_QA,
            need_retrieval=False,
            resolved_query=ctx.user_message,
            direct_answer=rule_answer,
        )
        # 应用识别状态，更新 graph state
        apply_intent_analysis(ctx, analysis)
        return state
		# 无知识库，没法检索，也是闲聊
    if ctx.conversation.knowledge_base_id is None:
        analysis = IntentAnalysis(
            kind=IntentKind.CHITCHAT,
            need_retrieval=False,
            resolved_query=ctx.user_message,
        )
        apply_intent_analysis(ctx, analysis)
        return state

   # Flash + few-shot：单次结构化输出意图、改写问句及可选拆分/澄清要点。
    deps = state["deps"]
    llm = deps.get("llm_fast") or deps["llm"]
    analysis = await analyze_intent(
        llm,
        user_message=ctx.user_message,
        history=ctx.history,
    )
    apply_intent_analysis(ctx, analysis)
    source = (
        "llm_fallback"
        if analysis.detail.get("reasoning") == "intent_llm_fallback"
        else "llm_fast"
    )
    writer(_intent_done_event(ctx, source=source))
    return state
```

## retrieval

项目使用的是双路融合检索，ES 关键词检索加上 Milvus 向量检索，再对召回结果进行过滤、重排。搜索功能都是调用工具，后面展开讲相关工具，这里大致看下过滤重排逻辑吧。

```python
# 检索方法
async def retrieve(
    self,
    query: str,
    *,
    tenant_id: UUID,
    kb_id: UUID,
    top_k: int = 20,
    top_n: int = 8,
) -> RetrievalResult:
    """多通道检索，融合得分，重排序，返回检索结果。"""
    # 向量检索
    vector = VectorChannelRetriever(
        embedding=self._embedding_provider,
        milvus_repo=self._milvus_repo,
        tenant_id=tenant_id,
        kb_id=kb_id,
        top_k=top_k,
    )
    # 关键词，使用 ik 分词
    bm25 = BM25ChannelRetriever(
        es_repo=self._es_repo,
        tenant_id=tenant_id,
        kb_id=kb_id,
        top_k=top_k,
    )

    latency: dict[str, float] = {}
    # 检索结果
    channel_docs = await asyncio.gather(
        _timed_ainvoke(vector, query, latency, "vector"),
        _timed_ainvoke(bm25, query, latency, "bm25"),
    )

    ensemble = EnsembleRetriever(
        retrievers=[vector, bm25],
        weights=[0.7, 0.3],
        c=_RRF_K,
        id_key="chunk_id",
    )
    # 按照不同权重，rrf 融合
    fused_docs = ensemble.weighted_reciprocal_rank(list(channel_docs))
    for doc in fused_docs:
        channels_hit = [
            name
            for name, docs in zip(_CHANNEL_NAMES, channel_docs, strict=True)
            if any(
                d.metadata.get("chunk_id") == doc.metadata["chunk_id"] for d in docs
            )
        ]
        doc.metadata["channels"] = channels_hit

    rerank_start = time.perf_counter()
    # 重排
    compressor = RerankCompressor(
        rerank_provider=self._rerank_provider, top_n=top_n
    )
    reranked_docs = await compressor.acompress_documents(fused_docs, query)
    latency["rerank"] = (time.perf_counter() - rerank_start) * 1000

    expand_start = time.perf_counter()
    # 检索使用子块，对话使用父块，根据子串找到父块
    expanded_docs = await self._expand_to_parents(
        reranked_docs, tenant_id=tenant_id
    )
    latency["expand"] = (time.perf_counter() - expand_start) * 1000

    return RetrievalResult(
        query=query,
        chunks=tuple(_document_to_fused(doc) for doc in expanded_docs),
        channel_hits={
            name: tuple(document_to_chunk(doc) for doc in docs)
            for name, docs in zip(_CHANNEL_NAMES, channel_docs, strict=True)
        },
        latency_ms=latency,
    )
```

这是单个问题检索方法，意图识别后，可能会拆分子问题，子问题查询后，还需要针对不同问题进行融合。逻辑差不多，就是每个子问题结果遍历一边，同个 chunk 取 rerank 最高分，所有 chunk 降序排列。

```python
async def generate_node(state: ChatState, writer: StreamWriter) -> ChatState:
    """生成助手回答：规则短路直出，或按意图选择 RAG/闲聊提示词流式生成。"""
    ctx = state["ctx"]
    writer(StageStatusEvent(name="generate", status="start"))

    if ctx.skip_generate and ctx.answer is not None:
        writer(DeltaEvent(content=ctx.answer))
        writer(
            StageStatusEvent(
                name="generate",
                status="done",
                detail={"answer_chars": len(ctx.answer), "skipped": True},
            )
        )
        return state

    deps = state["deps"]
    settings: Settings = deps["settings"]
    counter = get_token_counter(settings)
    budget = ContextBudget.from_settings(settings)
    question = ctx.resolved_query or ctx.user_message
    is_chitchat = ctx.intent == IntentKind.CHITCHAT.value
    # 打包上下文
    packed = pack_prompt_parts(
        counter=counter,
        budget=budget,
        summary=ctx.summary,
        memories=ctx.memories,
        retrieval=ctx.retrieval,
        history=ctx.history,
        question=question,
        skills=ctx.skills,
        tool_results=ctx.tool_results,
        include_retrieval=not is_chitchat,
    )
    ctx.overflow_history = list(packed.overflow_history)
    ctx.context_usage = packed.usage
    ctx.history = packed.history

    llm = deps["llm"]
    if is_chitchat:
        chain = _CHITCHAT_PROMPT | llm
        prompt_vars = {
            "skills": packed.skills,
            "summary": packed.summary,
            "memories": packed.memories,
            "history": packed.history,
            "question": packed.question,
        }
    else:
        chain = _GENERATE_PROMPT | llm
        prompt_vars = {
            "skills": packed.skills,
            "summary": packed.summary,
            "memories": packed.memories,
            "context": packed.context,
            "tool_results": packed.tool_results,
            "history": packed.history,
            "question": packed.question,
        }

    parts: list[str] = []
    async for chunk in chain.astream(prompt_vars):
        if (
            isinstance(chunk, AIMessageChunk)
            and isinstance(chunk.content, str)
            and chunk.content
        ):
            parts.append(chunk.content)
            # 推送前端
            writer(DeltaEvent(content=chunk.content))
		# 回答完成，拼接结果
    ctx.answer = "".join(parts)
    writer(
        StageStatusEvent(
            name="generate",
            status="done",
            detail={
                "answer_chars": len(ctx.answer),
                "context_tokens": packed.usage.total,
                "history_dropped": packed.usage.history_dropped,
                "retrieval_dropped": packed.usage.retrieval_dropped,
            },
        )
    )
    return state
```

## persist

回答完成后，需要讲对话持久化入库。完整对话保存在 PG 中，同时会更新记忆，总结对话。

```python
async def persist_node(state: ChatState, writer: StreamWriter) -> ChatState:
    """将本轮问答写入 PG/Redis，并在溢出时折叠对话摘要。"""
    ctx = state["ctx"]
    if ctx.answer is None:
        writer(StageStatusEvent(name="persist", status="skip"))
        return state

    deps = state["deps"]
    writer(StageStatusEvent(name="persist", status="start"))
    answer = ctx.answer
    # 更新对话历史，两条，用户和助手回复
    await deps["conversation_repo"].append_message(
        conversation_id=ctx.conversation.id,
        role="user",
        content=ctx.user_message,
        trace_id=ctx.trace_id,
    )
    assistant_message = await deps["conversation_repo"].append_message(
        conversation_id=ctx.conversation.id,
        role="assistant",
        content=answer,
        trace_id=ctx.trace_id,
        retrieval_detail=_retrieval_snapshot(ctx.retrieval),
    )
    ctx.assistant_message_id = assistant_message.id
    # 更新短期记忆，滑动窗口，超出头部自动出列
    await deps["short_memory"].add_user_message(ctx.conversation.id, ctx.user_message)
    await deps["short_memory"].add_ai_message(ctx.conversation.id, answer)

    summary_updated = False
    # 上下文中对话历史做了裁切，需要更新对话总结，保留重点
    if ctx.overflow_history:
        settings: Settings = deps["settings"]
        counter = get_token_counter(settings)
        budget = ContextBudget.from_settings(settings)
        fast_llm = deps.get("llm_fast") or deps["llm"]
        try:
            new_summary = await merge_conversation_summary(
                fast_llm,
                prior_summary=ctx.summary,
                overflow=ctx.overflow_history,
                counter=counter,
                max_tokens=budget.summary,
            )
            await deps["conversation_repo"].update_summary(
                ctx.conversation, summary=new_summary or None
            )
            ctx.summary = new_summary or None
            summary_updated = True
        except Exception:
            import structlog

            structlog.get_logger().warning(
                "conversation_summary_update_failed", exc_info=True
            )
		# 长期记忆更新，用户肖像
    long_memory = deps.get("long_memory")
    if long_memory is not None:
        await long_memory.remember_exchange(ctx.actor.id, ctx.user_message, answer)
    writer(
        StageStatusEvent(
            name="persist",
            status="done",
            detail={"summary_updated": summary_updated},
        )
    )
    return state

```

DeepSeek 没有长期记忆，每次对话都是新的。对比使用 ChatGPT，会明显感受到差异。ChatGPT 你可以开多个对话，相似内容他都会记得，能根据以前的问答给出建议。专注进行一个任务，又想提问不同方面时，长期记忆使得你不必重复叙述内容。比如，你问他如何经营自己的社交媒体账号，给出自己的基本情况，让他作出大致规划。看过大致计划后，你想对某个细节如何落地展开追问，重开了一个对话，ChatGPT 会在给出实践方式后，结合之前的方案给出建议。这样你就不必在每次问答后，重复询问，在我这个预算下怎么选择。为什么要展开询问？计划中是有取舍的，计划只能给出适合大众的选项，在你追问细节后，更能找到适合自己的。这样在你不断追问下，整个计划会更丰富可行。本地开发中，ide 如果没有长期记忆，可以使用 skill。开发前，首先创建 skill，写入 template 和 规范。每次做了调整，更新 skill。这样你会发现，代码的补全效果越来越好。

```python
async def title_node(state: ChatState, writer: StreamWriter) -> ChatState:

    ctx = state["ctx"]
    if not (ctx.is_first_exchange and ctx.answer):
        writer(StageStatusEvent(name="title", status="skip"))
        return state

    deps = state["deps"]
    writer(StageStatusEvent(name="title", status="start"))
    title: str | None = None
    try:
      	# 总结使用小模型即可
        llm = deps.get("llm_fast") or deps["llm"]
        chain = _TITLE_PROMPT | llm.bind(temperature=0.2) | StrOutputParser()
        raw = await chain.ainvoke(
            {
                "question": ctx.user_message[:200],
                "answer": (ctx.answer or "")[:300],
            }
        )
        title = sanitize_title(raw) or sanitize_title(ctx.user_message) or None
    except Exception:
        structlog.get_logger().warning("conversation_title_llm_failed", exc_info=True)
        title = sanitize_title(ctx.user_message) or None

    if title:
        await deps["conversation_repo"].update_title(ctx.conversation, title=title)
        ctx.title = title
        writer(TitleEvent(title=title))
    writer(StageStatusEvent(name="title", status="done", detail={"title": title}))
    return state


def sanitize_title(raw: str) -> str:
    """清洗标题噪声并截断到侧边栏展示长度。"""
    text = raw.strip().splitlines()[0].strip() if raw.strip() else ""
    text = text.strip("「」『』\"'" "''【】[]()（）")  # noqa: B005
    text = re.sub(r"\s+", "", text)
    if not text or text == _DEFAULT_TITLE:
        return ""
    return text[:_TITLE_MAX_LEN]
```
